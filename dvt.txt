
#include <stdio.h>

int main() {
    int d[3][3] = {
        {0, 2, 7},
        {2, 0, 1},
        {7, 1, 0}
    };
    int i, j, k;

    for (k = 0; k < 3; k++)
        for (i = 0; i < 3; i++)
            for (j = 0; j < 3; j++)
                if (d[i][k] + d[k][j] < d[i][j])
                    d[i][j] = d[i][k] + d[k][j];

    printf("Distance Vector Table:\n");
    for (i = 0; i < 3; i++) {
        for (j = 0; j < 3; j++)
            printf("%d ", d[i][j]);
        printf("\n");
    }

    return 0;
}
