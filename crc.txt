#include <stdio.h>
#include <string.h>

int main() {
    char data[20], gen[10], temp[30];
    int i, j, n, m;

    printf("Enter data bits: ");
    scanf("%s", data);

    printf("Enter generator bits: ");
    scanf("%s", gen);

    n = strlen(data);
    m = strlen(gen);

    strcpy(temp, data);

    for (i = 0; i < m - 1; i++)
        temp[n + i] = '0';
    temp[n + m - 1] = '\0';

    for (i = 0; i < n; i++) {
        if (temp[i] == '1') {
            for (j = 0; j < m; j++) {
                temp[i + j] =
                    (temp[i + j] == gen[j]) ? '0' : '1';
            }
        }
    }

    printf("CRC remainder: ");
    for (i = n; i < n + m - 1; i++)
        printf("%c", temp[i]);

    printf("\n");
    return 0;
}
