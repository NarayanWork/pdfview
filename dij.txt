
#include <stdio.h>

int main() {
    int cost[3][3] = {
        {0, 2, 7},
        {2, 0, 1},
        {7, 1, 0}
    };
    int d[3] = {0, 999, 999};
    int v[3] = {0}, i, j, u, min;

    for (i = 0; i < 3; i++) {
        u = -1;
        min = 999;

        for (j = 0; j < 3; j++)
            if (!v[j] && d[j] < min)
                min = d[j], u = j;

        if (u == -1) break;
        v[u] = 1;

        for (j = 0; j < 3; j++)
            if (!v[j] && d[u] + cost[u][j] < d[j])
                d[j] = d[u] + cost[u][j];
    }

    printf("Shortest distances from Router 0:\n");
    for (i = 0; i < 3; i++)
        printf("Router %d = %d\n", i, d[i]);

    return 0;
}
